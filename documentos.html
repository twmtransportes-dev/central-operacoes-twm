/* ============================================================ TWM Apps Script — MÓDULO DOCUMENTAÇÃO ------------------------------------------------------------ PROJETO SEPARADO — não mexe nos outros scripts. COMO USAR: 1) Crie uma planilha Google nova e vazia, chame de "TWM Documentos". Copie o ID dela (trecho da URL entre /d/ e /edit) e cole na constante DOCUMENTOS_ID abaixo. 2) Vá em script.google.com/home → "Novo projeto", apague o conteúdo padrão e cole este arquivo inteiro. 3) Implantar → Nova implantação → Aplicativo da Web - Executar como: Eu - Quem pode acessar: Qualquer pessoa Implantar → autorizar. (Nas próximas alterações: Gerenciar implantações → Editar → Nova versão — assim a URL não muda.) 4) Copie a URL /exec e cole em documentos.html na constante SCRIPT_URL. Os arquivos ficam salvos no Google Drive, numa pasta raiz "TWM DataLink - Documentos" (criada automaticamente na primeira vez), organizada em subpastas por frota/motorista. ============================================================ */ // Planilha Frota (a aba DOCUMENTOS é criada automaticamente) var DOCUMENTOS_ID = '1Q1XHvwuho8nNFAUMd3XcZx70njgU7kKatVnbk45X5Xg'; var ABA_DOCUMENTOS = 'DOCUMENTOS'; var PASTA_RAIZ_DOCS = 'Documentos Cavalos e Carretas'; // nome da pasta principal no Drive // opcional: ID da pasta (mais seguro que o nome). Se vazio, procura pelo nome. var PASTA_RAIZ_ID = ''; var TIPOS_VALIDOS = ['CRLV','CIV','CIPP','CNH','CTE','NOTA_CARGA','ROTOGRAMA','OUTRO']; function out(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var action = e.parameter.action || ''; try { if (action === 'getDocumentos') return getDocumentos(e); if (action === 'ping') return out({ ok: true, msg: 'Documentos Script ativo!' }); return out({ ok: true, msg: 'Documentos Script ativo!' }); } catch (err) { return out({ ok: false, msg: 'Erro: ' + err.toString() }); } } function doPost(e) { try { var body = JSON.parse(e.postData.contents || '{}'); var action = body.action || ''; if (action === 'uploadDocumento') return uploadDocumento(body); if (action === 'excluirDocumento') return excluirDocumento(body); return out({ ok: false, msg: 'Ação inválida.' }); } catch (err) { return out({ ok: false, msg: 'Erro: ' + err.toString() }); } } function _sheetDocumentos() { var ss = SpreadsheetApp.openById(DOCUMENTOS_ID); var sh = ss.getSheetByName(ABA_DOCUMENTOS); if (!sh) { sh = ss.insertSheet(ABA_DOCUMENTOS); var cab = ['ID','TIPO','VINCULO_TIPO','VINCULO_VALOR','NOME_ARQUIVO','URL_DRIVE','DATA_VALIDADE','DATA_UPLOAD']; sh.appendRow(cab); var hh = sh.getRange(1,1,1,cab.length); hh.setBackground('#1a1a2e'); hh.setFontColor('#FFFFFF'); hh.setFontWeight('bold'); hh.setFontSize(10); sh.setFrozenRows(1); } return sh; } // Compara nomes ignorando maiúsculas, acentos e espaços duplos function _normNome(n) { return String(n || '').normalize('NFD').replace(/[\u0300-\u036f]/g, '') .replace(/\s+/g, ' ').trim().toUpperCase(); } function _pastaRaiz() { if (PASTA_RAIZ_ID) return DriveApp.getFolderById(PASTA_RAIZ_ID); var alvos = [_normNome(PASTA_RAIZ_DOCS), _normNome('TWM DataLink - Documentos')]; var cand = DriveApp.searchFolders('trashed = false and (title contains "Cavalos" or title contains "TWM DataLink")'); var achadas = {}; while (cand.hasNext()) { var f = cand.next(); var n = _normNome(f.getName()); if (alvos.indexOf(n) >= 0 && !achadas[n]) achadas[n] = f; } for (var i = 0; i < alvos.length; i++) if (achadas[alvos[i]]) return achadas[alvos[i]]; return DriveApp.createFolder(PASTA_RAIZ_DOCS); } // Acha a subpasta da frota/motorista. Aceita "VP-183", "VP-183 RML-0I14" ou "FROTA - VP-183". function _subPasta(vinculoTipo, vinculoValor) { var raiz = _pastaRaiz(); var v = _normNome(vinculoValor); var subs = raiz.getFolders(); while (subs.hasNext()) { var f = subs.next(); var n = _normNome(f.getName()).replace(/^(FROTA|MOTORISTA)\s*-\s*/, ''); if (n === v || n.indexOf(v + ' ') === 0) return f; } return raiz.createFolder(vinculoTipo === 'FROTA' ? vinculoValor : ('MOTORISTA - ' + vinculoValor)); } // Normaliza validade para "aaaa-mm-dd" (o Sheets converte texto de data em Date) function _fmtValidade(v) { if (!v) return ''; if (Object.prototype.toString.call(v) === '[object Date]') { if (isNaN(v.getTime())) return ''; return Utilities.formatDate(v, Session.getScriptTimeZone(), 'yyyy-MM-dd'); } var s = String(v).trim(); var m = s.match(/^(\d{2})\/(\d{2})\/(\d{4})$/); // dd/mm/aaaa → aaaa-mm-dd if (m) return m[3] + '-' + m[2] + '-' + m[1]; return s; } /* ── UPLOAD (POST — arquivo em base64) ──────────────────────── Espera no corpo (JSON): tipo, vinculoTipo (FROTA|MOTORISTA), vinculoValor, validade (opcional, aaaa-mm-dd), nomeArquivo, mimeType, base64 (sem o prefixo data:...;base64,) ------------------------------------------------------------- */ function uploadDocumento(p) { var tipo = String(p.tipo || '').trim().toUpperCase(); var vinculoTipo = String(p.vinculoTipo || '').trim().toUpperCase(); var vinculoValor = String(p.vinculoValor || '').trim().toUpperCase(); var validade = String(p.validade || '').trim(); var nomeArquivo = p.nomeArquivo || ('documento_' + Date.now()); var mimeType = p.mimeType || 'application/octet-stream'; var base64 = p.base64 || ''; if (!tipo || !vinculoTipo || !vinculoValor || !base64) { return out({ ok: false, msg: 'Dados incompletos para o upload.' }); } if (TIPOS_VALIDOS.indexOf(tipo) < 0) tipo = 'OUTRO'; try { var bytes = Utilities.base64Decode(base64); var blob = Utilities.newBlob(bytes, mimeType, nomeArquivo); var pasta = _subPasta(vinculoTipo, vinculoValor); var arquivo = pasta.createFile(blob); arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); var id = Utilities.getUuid(); var sh = _sheetDocumentos(); sh.appendRow([id, tipo, vinculoTipo, vinculoValor, nomeArquivo, arquivo.getUrl(), validade, new Date().toLocaleString('pt-BR')]); return out({ ok: true, id: id, url: arquivo.getUrl() }); } catch (err) { return out({ ok: false, msg: 'Erro ao salvar no Drive: ' + err.toString() }); } } /* ── EXCLUIR (POST) ───────────────────────────────────────── */ function excluirDocumento(p) { var id = String(p.id || ''); var sh = _sheetDocumentos(); var dados = sh.getDataRange().getValues(); for (var i = 1; i < dados.length; i++) { if (String(dados[i][0]) === id) { try { var url = String(dados[i][5] || ''); var fid = url.match(/[-\w]{25,}/); if (fid) DriveApp.getFileById(fid[0]).setTrashed(true); } catch (e) {} sh.deleteRow(i + 1); return out({ ok: true }); } } return out({ ok: false, msg: 'Documento não encontrado.' }); } /* ── SINCRONIZAR DRIVE ──────────────────────────────────────── Arquivos colocados direto nas pastas do Drive são registrados na planilha automaticamente. Padrão de pastas (dentro de "TWM DataLink - Documentos"): "FROTA - VP-183" → vinculado à frota VP-183 "MOTORISTA - JOAO" → vinculado ao motorista JOAO "VP-183" (sem prefixo) → tratado como frota Tipo: pelo início do nome do arquivo (CRLV, CIV, CIPP, CNH, CTE, NOTA, ROTOGRAMA). Se não reconhecer → OUTRO. Validade (opcional): data no nome, ex: "CRLV VP-183 31-12-2026.pdf" ------------------------------------------------------------- */ // Arquivos de sistema que o Windows/Mac colocam nas pastas e não são documentos function _ehLixo(nome) { var n = String(nome || '').trim().toLowerCase(); if (!n) return true; if (n === 'thumbs.db' || n === 'desktop.ini' || n === 'ehthumbs.db') return true; if (n.charAt(0) === '.') return true; // .DS_Store, ._arquivo, etc. if (n.indexOf('~$') === 0) return true; // temporário do Office if (/\.(db|ini|tmp|lnk|url)$/.test(n)) return true; return false; } function _fileIdDaUrl(url) { var m = String(url || '').match(/[-\w]{25,}/); return m ? m[0] : ''; } function _tipoPeloNome(nome) { var n = String(nome || '').normalize('NFD').replace(/[̀-ͯ]/g, '').toUpperCase().trim(); if (n.indexOf('CRLV') === 0) return 'CRLV'; if (n.indexOf('CIPP') === 0) return 'CIPP'; if (n.indexOf('CIV') === 0) return 'CIV'; if (n.indexOf('CNH') === 0) return 'CNH'; if (/^CT-?E/.test(n)) return 'CTE'; if (n.indexOf('NOTA') === 0 || /^NF[\s_-]/.test(n)) return 'NOTA_CARGA'; if (n.indexOf('ROTOGRAMA') === 0) return 'ROTOGRAMA'; return 'OUTRO'; } function _validadePeloNome(nome) { var m = String(nome || '').match(/(\d{2})[-._](\d{2})[-._](\d{4})/); return m ? (m[3] + '-' + m[2] + '-' + m[1]) : ''; } function _vinculoDaPasta(nomePasta) { var n = String(nomePasta || '').trim().toUpperCase(); var m = n.match(/^(FROTA|MOTORISTA)\s*-\s*(.+)$/); if (m) return { tipo: m[1], valor: m[2].trim() }; return { tipo: 'FROTA', valor: n }; } // Registra os arquivos de UMA pasta que ainda não estão na planilha function _sincronizarPasta(pasta, sh, idsExistentes) { var vinc = _vinculoDaPasta(pasta.getName()); var novos = 0; var arquivos = pasta.getFiles(); while (arquivos.hasNext()) { var arq = arquivos.next(); if (arq.isTrashed()) continue; var fid = arq.getId(); if (idsExistentes[fid]) continue; if (_ehLixo(arq.getName())) continue; try { arq.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {} var nome = arq.getName(); sh.appendRow([Utilities.getUuid(), _tipoPeloNome(nome), vinc.tipo, vinc.valor, nome, arq.getUrl(), _validadePeloNome(nome), new Date().toLocaleString('pt-BR')]); idsExistentes[fid] = true; novos++; } return novos; } // filtro opcional: {tipo:'FROTA'|'MOTORISTA', valor:'VP-183'} → só pastas que batem function sincronizarDrive(filtro) { var sh = _sheetDocumentos(); var dados = sh.getDataRange().getValues(); // Limpa linhas de arquivos de sistema registradas antes (ex: Thumbs.db) for (var j = dados.length - 1; j >= 1; j--) { if (dados[j][0] && _ehLixo(dados[j][4])) sh.deleteRow(j + 1); } dados = sh.getDataRange().getValues(); var ids = {}; for (var i = 1; i < dados.length; i++) { var fid = _fileIdDaUrl(dados[i][5]); if (fid) ids[fid] = true; } var novos = 0; var subs = _pastaRaiz().getFolders(); while (subs.hasNext()) { var pasta = subs.next(); if (filtro) { var v = _vinculoDaPasta(pasta.getName()); if (v.tipo !== filtro.tipo || v.valor.indexOf(filtro.valor) < 0) continue; } novos += _sincronizarPasta(pasta, sh, ids); } return novos; } /* ── BUSCAR (GET) ───────────────────────────────────────────── Se "frota" for informado, retorna só documentos vinculados a essa frota. Se "motorista" for informado, só os vinculados a esse motorista. Sem nenhum dos dois, retorna todos (usado na aba "Gerenciar", só para admin). ------------------------------------------------------------- */ function getDocumentos(e) { var p = e.parameter || {}; var frota = String(p.frota || '').trim().toUpperCase(); var motorista = String(p.motorista || '').trim().toUpperCase(); // Sincroniza com o Drive antes de listar: // - Gerenciar (sync=1): todas as pastas // - Busca por frota/motorista: só a pasta correspondente (rápido) var novos = 0; try { if (p.sync === '1') novos = sincronizarDrive(null); else if (frota) novos = sincronizarDrive({ tipo: 'FROTA', valor: frota }); else if (motorista) novos = sincronizarDrive({ tipo: 'MOTORISTA', valor: motorista }); } catch (errSync) { Logger.log('Erro sync Drive: ' + errSync); } var sh = _sheetDocumentos(); var dados = sh.getDataRange().getValues(); var lista = []; for (var i = 1; i < dados.length; i++) { var r = dados[i]; if (!r[0]) continue; if (_ehLixo(r[4])) continue; var item = { id: String(r[0]), tipo: String(r[1] || ''), vinculoTipo: String(r[2] || ''), vinculoValor: String(r[3] || ''), nomeArquivo: String(r[4] || ''), url: String(r[5] || ''), validade: _fmtValidade(r[6]), dataUpload: String(r[7] || '') }; if (frota) { if (item.vinculoTipo === 'FROTA' && item.vinculoValor.indexOf(frota) >= 0) lista.push(item); continue; } if (motorista) { if (item.vinculoTipo === 'MOTORISTA' && item.vinculoValor.indexOf(motorista) >= 0) lista.push(item); continue; } lista.push(item); // sem filtro: lista tudo } return out({ ok: true, documentos: lista, novos: novos }); }
